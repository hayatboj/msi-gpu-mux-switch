find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network DBus Svg)
qt_standard_project_setup()
set(CMAKE_AUTORCC ON)

add_library(msi-mux-ui-core STATIC status.cpp strings.cpp backend.cpp selection.cpp modehero.cpp poweractions.cpp desktopintegration.cpp aboutdialog.cpp)
target_link_libraries(msi-mux-ui-core PUBLIC Qt6::Widgets Qt6::Network Qt6::DBus)
target_include_directories(msi-mux-ui-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(msi-mux-ui-core PUBLIC cxx_std_17)
target_compile_definitions(msi-mux-ui-core PUBLIC MSI_MUX_VERSION="${PROJECT_VERSION}")
target_compile_options(msi-mux-ui-core PRIVATE -Wall -Wextra -Wpedantic)

qt_add_executable(msi-mux-tray main.cpp window.cpp resources.qrc)
target_link_libraries(msi-mux-tray PRIVATE msi-mux-ui-core Qt6::Svg)
target_compile_options(msi-mux-tray PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS msi-mux-tray RUNTIME DESTINATION bin)
install(FILES icons/msi-mux.svg DESTINATION share/icons/hicolor/scalable/apps RENAME org.hayatboj.msimux.svg)

if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    qt_add_executable(msi-mux-ui-tests tests/test_ui.cpp window.cpp resources.qrc)
    target_link_libraries(msi-mux-ui-tests PRIVATE msi-mux-ui-core Qt6::Test)
    target_compile_definitions(msi-mux-ui-tests PRIVATE MSI_MUX_TESTING)
    add_test(NAME msi-mux-ui-tests COMMAND msi-mux-ui-tests)
    set_tests_properties(msi-mux-ui-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    foreach(suite IN ITEMS power desktopintegration about selection)
        qt_add_executable(msi-mux-${suite}-tests tests/test_${suite}.cpp resources.qrc)
        target_link_libraries(msi-mux-${suite}-tests PRIVATE msi-mux-ui-core Qt6::Test)
        add_test(NAME msi-mux-${suite}-tests COMMAND msi-mux-${suite}-tests)
        set_tests_properties(msi-mux-${suite}-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    endforeach()
    add_test(NAME msi-mux-launch-tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_launch.py" "$<TARGET_FILE:msi-mux-tray>")
    qt_add_executable(msi-mux-instance-tests tests/test_instance.cpp)
    target_link_libraries(msi-mux-instance-tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    target_compile_definitions(msi-mux-instance-tests PRIVATE MSI_MUX_TRAY_TEST_BINARY="$<TARGET_FILE:msi-mux-tray>")
    add_dependencies(msi-mux-instance-tests msi-mux-tray)
    add_test(NAME msi-mux-instance-tests COMMAND msi-mux-instance-tests)
    set_tests_properties(msi-mux-launch-tests msi-mux-instance-tests PROPERTIES TIMEOUT 60)
endif()
